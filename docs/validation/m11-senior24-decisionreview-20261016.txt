Exact grading production-path execution

An ephemeral Go harness was created in a temporary directory under the
repository and removed in a finally block. It loaded the embedded published
spec, bound the trusted reference, called Spec.Request(reference, true), ran
that request once using runner.New().Execute (rootless Podman; pinned Go

true). The private request combines all public and private vectors.

Harness source:

  package main
  import (
    "context"
    "fmt"
    "os"
    "academy/internal/grading"
    "academy/internal/runner"
  )
  func main() {
    specs, err := grading.LoadFor([]string{"senior-24-medium"}); check(err)
    spec := specs["senior-24-medium"]
    reference, err := os.ReadFile("internal/content/solutions/senior-24-medium/main.go"); check(err)
    check(spec.BindReference(string(reference)))
    request, err := spec.Request(map[string]string{"main.go": string(reference)}, true); check(err)
    fmt.Printf("spec=%s mode=%s vectors=16 args-bytes=%d\n", spec.ExerciseID(), request.Mode, len(request.Args[0]))
    result, err := runner.New().Execute(context.Background(), request); check(err)
    checks := spec.Check(result, true)
    passed := 0
    for _, item := range checks {
      if item.Passed { passed++ }
      fmt.Printf("%s: %t\n", item.Name, item.Passed)
    }
    fmt.Printf("runner-status=%s exit-code=%d stderr-bytes=%d checks=%d passed=%d\n",
      result.Status, result.ExitCode, len(result.Stderr), len(checks), passed)
    if passed != 16 || result.Status != "ok" || result.ExitCode != 0 || result.Stderr != "" { os.Exit(1) }
  }
  func check(err error) { if err != nil { fmt.Fprintln(os.Stderr, err); os.Exit(2) } }
